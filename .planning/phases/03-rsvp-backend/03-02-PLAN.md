    with the following exact content:

    ```js
    // functions/api/rsvp.js
    // Cloudflare Pages Function — auto-routes to /api/rsvp
    // Auth is enforced upstream by functions/_middleware.js — do not add auth logic here.

    export async function onRequestPost(context) {
      const { request, env } = context;

      // Parse JSON body — return 400 on malformed input
      let body;
      try {
        body = await request.json();
      } catch {
        return Response.json({ ok: false, error: 'Invalid JSON' }, { status: 400 });
      }

      const { name, contact, attendees } = body;

      // Server-side validation (frontend validates too, but Worker is the last gate)
      if (!name || typeof name !== 'string' || name.trim().length === 0) {
        return Response.json({ ok: false, error: 'Name required' }, { status: 400 });
      }

      const attendeesInt = parseInt(attendees, 10);
      if (isNaN(attendeesInt) || attendeesInt < 1) {
        return Response.json({ ok: false, error: 'Invalid attendees' }, { status: 400 });
      }

      // Coerce empty contact string to null — contact_method column is nullable (D-10)
      const contactMethod = (contact && contact.length > 0) ? contact : null;

      // Generate unique id — crypto.randomUUID() is native in Workers runtime, no import needed (D-09)
      const id = crypto.randomUUID();
      const now = new Date().toISOString();

      // Insert new row — every submission is a new record, no upsert (D-11)
      try {
        await env.DB.prepare(
          'INSERT INTO rsvps (id, name, contact_method, attendees, submitted_at) VALUES (?, ?, ?, ?, ?)'
        ).bind(id, name.trim(), contactMethod, attendeesInt, now).run();
      } catch (err) {
        // Do not expose internal error details to the client
        return Response.json({ ok: false, error: 'Database error' }, { status: 500 });
      }

      return Response.json({ ok: true }, { status: 200 });
    }
    ```

    Key implementation notes:
    - Export name MUST be `onRequestPost` (not `onRequest`) — Pages Functions method-specific
      export; other HTTP methods automatically return 405 with no extra code required.
    - `context.env.DB` references the D1 binding named "DB" in wrangler.jsonc.
    - `crypto.randomUUID()` is a global — no import needed in Cloudflare Workers runtime.
    - `name.trim()` is applied both in validation and when storing — prevents whitespace-only names.
    - Error response body is generic (`'Database error'`) — the raw D1 error is not forwarded.
    - The `catch` block on `request.json()` uses empty `catch` (no parameter) — compatible with
      the project's plain JS approach and Workers runtime.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Internet → /api/rsvp | Untrusted JSON body crosses here; auth middleware runs first but body content is still attacker-controlled |
| Worker → D1 | Worker issues SQL via .prepare().bind() — parameterized, not string concatenation |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-02-01 | Tampering | request.json() body | mitigate | try/catch on request.json() returns 400 on malformed input; validated fields before use |
| T-03-02-02 | Tampering | SQL injection via name/contact | mitigate | D1 `.prepare().bind()` uses parameterized queries — user input never interpolated into SQL string |
| T-03-02-03 | Information Disclosure | DB error details leaked to client | mitigate | catch block returns generic `'Database error'` string — raw D1 error not forwarded |
| T-03-02-04 | Denial of Service | submission flood | accept | personal event site, low-value target; D1 free tier limits provide a natural cap; no rate limiting planned |
| T-03-02-05 | Elevation of Privilege | bypass auth to reach /api/rsvp | accept | _middleware.js enforces Basic Auth on ALL routes before any function runs; no additional auth logic needed in Worker |
</threat_model>
